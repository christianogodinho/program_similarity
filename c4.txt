import math

def f(x):
    return x**2

def integral_trapezios(a, b, n):
    h = (b - a) / n

    soma = (f(a) + f(b)) / 2

    for i in range(1, n):
        soma += f(a + i * h)

    return soma * h

a = 0
b = 1
n = 1000

resultado = integral_trapezios(a, b, n)

print(f"Integral de x² entre {a} e {b}: {resultado}")
